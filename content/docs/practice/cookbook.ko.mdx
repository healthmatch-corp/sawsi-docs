## 1. 새 DynamoModel 추가

```python title="main/model/post.py"
from sawsi.model.base import DynamoModel, sync
import api

@sync
class Post(DynamoModel):
    __dynamo__ = api.dynamo
    _table = 'post'
    _indexes = [
        ('author_id', 'crt'),         # 작성자별 글을 시간순
        ('category', 'crt'),           # 카테고리의 모든 글
    ]

    author_id: str
    title: str
    content: str
    category: str = 'general'
    is_published: bool = False
```

```python

```

## 2. 목록 페이지네이션

```python title="main/controller/post.py"
@controller
def list_my_posts(user, start_key: dict | None = None, **kwargs):
    if not user:
        raise errs.no_session
    items, end_key = Post.query(
        pk_field='author_id', pk_value=user.id,
        sk_field='crt', reverse=True,
        limit=20, start_key=start_key,
    )
    return {
        'items': [p.model_dump() for p in items],
        'cursor': end_key,
    }
```

클라이언트가 다음 요청에 `cursor`를 `start_key`로 다시 보내요.

```python title="main/service/social_auth/kakao_auth.py"
import requests

def verify_kakao_access_token(access_token):
    r = requests.get(
        'https://kapi.kakao.com/v2/user/me',
        headers={'Authorization': f'Bearer {access_token}'}, timeout=5,
    )
    if r.status_code != 200:
        return None
    data = r.json()
    return type('K', (), {
        'sub': str(data['id']),
        'name': data.get('properties', {}).get('nickname'),
        'email': data.get('kakao_account', {}).get('email'),
    })()
```

## 4. SQS 백그라운드 작업

```python title="main/service/post_service.py"
def publish_async(post_id):
    import json, api
    api.sqs.send_message(json.dumps({
        'type': 'queue_receiver.controller.post_publisher.run',
        'body': {'post_id': post_id},
    }))
```

```python title="queue_receiver/controller/post_publisher.py"
from sawsi.handler.controller_decorator import controller
from main.model.post import Post

@controller
def run(post_id: str, **kwargs):
    post = Post.get(post_id)
    if not post or post.is_published:
        return
    # 무거운 작업 — 팔로워에게 fan-out 등
    post.is_published = True
    post.update()
```

큐 워커(`queue_receiver/aws_handler.py`)가 `type`으로 메시지를 자동 라우팅해 줘요.

```python title="scheduler/controller/daily_digest.py"
from sawsi.handler.controller_decorator import controller
import api, time

@controller
def run(**kwargs):
    # 멱등하게 — EventBridge 이중 발사가 재실행하지 않게 락 사용
    key = time.strftime('daily_digest-%Y-%m-%d')
    try:
        with api.locking.lock(key, max_alive_ts=600, max_retries=1):
            _send_digests()
    except RuntimeError:
        pass   # 다른 호출에서 이미 실행 중
```

## 6. WebSocket 푸시

```python title="main/service/notification_service.py"
def notify_user_realtime(user_id, payload):
    import api, json
    from main.model.ws_connection import WsConnection
    for conn in WsConnection.generate(pk_field='user_id', pk_value=user_id):
        try:
            api.apigw.post_to_connection(
                data=json.dumps(payload),
                connection_id=conn.connection_id,
            )
        except Exception:
            conn.delete()   # 오래된 연결
```

## 8. 새 어드민 명령

```python title="admin/controller/user_admin.py"
from sawsi.handler.controller_decorator import controller
from main.model.user import User

@controller
def freeze_user(me, user_id: str, **kwargs):
    user = User.get(user_id)
    if not user:
        raise errs.no_such_user
    user.is_frozen = True
    user.update()
    api.firehose_log.put_record(json.dumps({
        'admin_id': me.id, 'action': 'freeze_user', 'user_id': user_id, 'ts': time.time(),
    }))
    return {'ok': True}
```

cmd를 적절한 `AdminGroup.allow_cmds`에 추가하거나 (어드민 UI 경유) 파괴적이면 `ROOT_ONLY_CMDS`에 추가하세요.

## 9. Bedrock + VectorKB

```python title="main/service/qa_service.py"
import api, json

def answer_question(question: str) -> str:
    candidates = api.vector_kb.search(question, top_k=5)
    context = '\n'.join(c['text'] for c in candidates)
    resp = api.bedrock.converse(
        model_id='anthropic.claude-haiku-4-5-20251001',
        system='Answer using only the provided context. If unknown, say so.',
        messages=[{
            'role': 'user',
            'content': [{'text': f'Context:\n{context}\n\nQuestion: {question}'}],
        }],
        max_tokens=512,
        temperature=0.0,
    )
    return resp['output']['message']['content'][0]['text']
```

## 12. 분산 락이 있는 임계 구역

```python title="main/controller/event.py"
@controller
def join_event(user, event_id: str, **kwargs):
    try:
        with api.locking.lock(f'join_event:{event_id}', max_alive_ts=10, max_retries=3, wait_time=1):
            event = Event.get(event_id)
            if event.participant_count >= event.max_participant_count:
                raise errs.exceed_max_participant_count
            event.participant_count += 1
            event.update()
            # 참여자 row도 작성 등
    except RuntimeError:
        raise errs.try_again_later
    return {'ok': True}
```

락이 두 동시 참여가 모두 용량 검사를 통과하는 것을 방지해 줘요.
